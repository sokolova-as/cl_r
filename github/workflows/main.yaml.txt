# Название рабочего процесса, которое будет отображаться на GitHub Actions
name: Lint and Format Check

# Триггеры: когда запускать этот процесс
on:
  push: # Запускать при каждой отправке (push) кода в репозиторий
    branches: [ main, master ] # Только для веток main и master (или укажите свои)
  pull_request: # Запускать при создании или обновлении pull request
    branches: [ main, master ] # Только для веток main и master

# Задачи (jobs), которые будут выполняться
jobs:
  lint: # Название задачи (может быть любым, например, 'check_code')
    # На какой операционной системе будет выполняться задача
    runs-on: ubuntu-latest # Рекомендуется использовать последнюю версию Ubuntusteps:
# Шаг 1: Скачивание кода из репозитория
- name: Checkout code
  uses: actions/checkout@v4 # Используем готовый "экшн" (действие) для скачивания кода

# Шаг 2: Настройка окружения Python
- name: Set up Python
  uses: actions/setup-python@v5
  with:
    python-version: '3.10' # Укажите версию Python, которую вы используете в своем проекте.
                           # Например, '3.9', '3.11' и т.д.
                           # Убедитесь, что эта версия установлена и работает локально.

# Шаг 3: Установка линтера Ruff
- name: Install Ruff
  run: pip install ruff # Команда для установки Ruff через менеджер пакетов pip

# Шаг 4: Запуск проверки Ruff
- name: Run Ruff check
  # Команда 'ruff check .' запускает Ruff для проверки всех файлов
  # в текущей директории ('.') и ее поддиректориях.
  # Ruff прочитает конфигурацию из pyproject.toml
  run: ruff check .
  # Если вы хотите, чтобы Ruff также автоматически исправлял найденные проблемы
  # (например, форматирование, расстановку импортов), раскомментируйте следующую строку
  # и закомментируйте 'ruff check .':
  # run: ruff check . --fix
  # Или, если вы хотите только форматирование (без проверок):
  # run: ruff format .
